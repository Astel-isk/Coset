const Q = 'C:/Users/PCuser/repository/hack-zAlo_zaukino/node_modules/qrcode-terminal/vendor/QRCode';
const QRCode = require(Q);
const L = require(Q + '/QRErrorCorrectLevel');
const qr = new QRCode(-1, L.M);
qr.addData('https://coset.astelisk.com');
qr.make();
const n = qr.getModuleCount(), q = 4, size = n + q * 2;
let d = '';
for (let r = 0; r < n; r++) for (let c = 0; c < n; c++) if (qr.isDark(r, c)) d += `M${c + q} ${r + q}h1v1h-1z`;
const svg = `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${size} ${size}" shape-rendering="crispEdges"><rect width="${size}" height="${size}" fill="#f3ecdc"/><path d="${d}" fill="#14202e"/></svg>`;
require('fs').writeFileSync(process.argv[2], svg);
console.log('modules', n, 'bytes', svg.length);
